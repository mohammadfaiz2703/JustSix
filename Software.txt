#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// Change address if needed (0x27 or 0x3F)
LiquidCrystal_I2C lcd(0x27, 16, 2);

#define SOIL_PIN A0

void setup() {
  Serial.begin(9600);

  // I2C start (SDA, SCL)
  Wire.begin(D2, D1);

  lcd.init();
  lcd.backlight();

  lcd.setCursor(0, 0);
  lcd.print("Smart Farming");
  delay(2000);
  lcd.clear();
}

void loop() {
  int soilValue = analogRead(SOIL_PIN);

  Serial.print("Soil Moisture: ");
  Serial.println(soilValue);

  lcd.setCursor(0, 0);
  lcd.print("Soil Value:");

  lcd.setCursor(0, 1);
  lcd.print(soilValue);
  lcd.print("     ");   // clears old digits

  delay(2000);
}
